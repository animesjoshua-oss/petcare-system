## CURRENT LOGIN IMPLEMENTATION (auth/login.php)

### AUTHENTICATION FLOW:

STEP 1: Email + Password Validation
  - User submits email and password
  - System checks if account exists
  - Validates password hash (BCrypt cost=12)

STEP 2: Account Status Checks
  - Check if account is locked (locked_until timestamp)
  - If locked and time not expired → Show countdown timer
  - If lock expired → Clear locked_until field
  - Check is_active flag (0 = pending approval, 1 = approved)
  - If inactive → Show "pending admin approval" message

STEP 3: Failed Login Tracking
  - On wrong password: Increment failed_login_attempts
  - After 5 failed attempts: Set locked_until = now + 5 minutes
  - On successful login: Reset failed_login_attempts to 0

STEP 4: Role-Based OTP Requirement
  - Pet owners (role: pet_owner) → REQUIRE OTP verification
  - Admin/Vet/Groomer → SKIP OTP, login directly
  
  For pet owners:
    - Generate 6-digit OTP
    - Hash OTP with password_hash()
    - Store in session: otp_code, otp_expires (10 minutes)
    - Store user ID in session: login_otp_user_id
    - Send OTP email via PHPMailer
    - Redirect to OTP verification page

  For staff (admin/vet/groomer):
    - Create session directly: $_SESSION['user_id']
    - Update last_login timestamp
    - Redirect by role to appropriate dashboard

STEP 5: OTP Verification (For Pet Owners Only)
  - User enters 6-digit OTP
  - System verifies with password_verify() against session hash
  - Check if OTP expired (10 minutes limit)
  - If valid: Login user, clear OTP from session
  - If invalid: Show error, allow retry
  - Resend OTP option available

STEP 6: Role-Based Redirect
  - pet_owner → /dashboard/owner/index.php
  - veterinarian → /dashboard/vet/index.php
  - groomer → /dashboard/groomer/index.php
  - admin/superadmin → /dashboard/admin/index.php

public function showLoginForm()

public function login(Request $request)
{
    // 1. Validate email + password
    $credentials = $request->validate([
        'email' => 'required|email',
        'password' => 'required|string',
    ]);

    // 2. Find user by email
    $user = User::where('email', $credentials['email'])->first();

    if (!$user) {
        return back()->withErrors(['email' => 'Invalid email or password.']);
    }

    // 3. Check account lockout
    if ($user->locked_until && now()->lt($user->locked_until)) {
        return back()->withErrors([
            'lockout' => 'Account locked. Try again in ' . 
                         now()->diffInSeconds($user->locked_until) . ' seconds.'
        ])->with('lockout_timestamp', $user->locked_until->timestamp);
    }

    // 4. Clear expired lockout
    if ($user->locked_until && now()->gte($user->locked_until)) {
        $user->update([
            'locked_until' => null,
            'failed_login_attempts' => 0
        ]);
    }

    // 5. Verify password
    if (!Hash::check($credentials['password'], $user->password_hash)) {
        // Increment failed attempts
        $attempts = $user->failed_login_attempts + 1;
        
        if ($attempts >= 5) {
            $user->update([
                'failed_login_attempts' => $attempts,
                'locked_until' => now()->addMinutes(5)
            ]);
            return back()->withErrors([
                'lockout' => 'Too many failed attempts. Account locked for 5 minutes.'
            ])->with('lockout_timestamp', $user->locked_until->timestamp);
        }
        
        $user->increment('failed_login_attempts');
        $remaining = 5 - $attempts;
        return back()->withErrors([
            'password' => "Invalid password. $remaining attempt(s) remaining."
        ]);
    }

    // 6. Check if account is active
    if (!$user->is_active) {
        return back()->withErrors([
            'inactive' => 'Your account is pending admin approval. 
                          You will be notified by email once approved.'
        ]);
    }

    // 7. Reset failed attempts on success
    $user->update(['failed_login_attempts' => 0, 'locked_until' => null]);

    // 8. Role-based OTP logic
    if ($user->role->name === 'pet_owner') {
        // Generate OTP
        $otp = str_pad(random_int(0, 999999), 6, '0', STR_PAD_LEFT);
        
        // Store in session
        session([
            'login_otp_user_id' => $user->id,
            'otp_code' => Hash::make($otp),
            'otp_expires' => now()->addMinutes(10)
        ]);
        
        // Send email
        Mail::to($user->email)->send(new LoginOtpMail($user, $otp));
        
        // Redirect to OTP page
        return redirect()->route('login.otp');
    }

    // 9. Login directly (admin/vet/groomer)
    Auth::login($user);
    $user->update(['last_login' => now()]);
    
    // 10. Redirect by role
    return $this->redirectByRole($user);
}

public function showOtpForm()
{
    if (!session()->has('login_otp_user_id')) {
        return redirect()->route('login');
    }
    
    $user = User::find(session('login_otp_user_id'));
    return view('auth.otp', compact('user'));
}

public function verifyOtp(Request $request)
{
    $request->validate(['otp' => 'required|digits:6']);
    
    // Check session
    if (!session()->has('otp_code') || !session()->has('otp_expires')) {
        return back()->withErrors(['otp' => 'OTP session expired.']);
    }
    
    // Check expiry
    if (now()->gt(session('otp_expires'))) {
        return back()->withErrors(['otp' => 'OTP expired. Please request a new one.']);
    }
    
    // Verify OTP
    if (!Hash::check($request->otp, session('otp_code'))) {
        return back()->withErrors(['otp' => 'Invalid OTP. Please try again.']);
    }
    
    // Login user
    $user = User::find(session('login_otp_user_id'));
    Auth::login($user);
    $user->update(['last_login' => now()]);
    
    // Clear OTP session
    session()->forget(['login_otp_user_id', 'otp_code', 'otp_expires']);
    
    // Redirect
    return $this->redirectByRole($user);
}

public function resendOtp()
{
    if (!session()->has('login_otp_user_id')) {
        return back()->withErrors(['error' => 'Session expired.']);
    }
    
    $user = User::find(session('login_otp_user_id'));
    
    // Generate new OTP
    $otp = str_pad(random_int(0, 999999), 6, '0', STR_PAD_LEFT);
    
    // Update session
    session([
        'otp_code' => Hash::make($otp),
        'otp_expires' => now()->addMinutes(10)
    ]);
    
    // Send email
    Mail::to($user->email)->send(new LoginOtpMail($user, $otp));
    
    return back()->with('success', 'New OTP sent to your email.');
}

protected function redirectByRole($user)
{
    return match($user->role->name) {
        'pet_owner' => redirect()->route('owner.dashboard'),
        'veterinarian' => redirect()->route('vet.dashboard'),
        'groomer' => redirect()->route('groomer.dashboard'),
        'admin', 'superadmin' => redirect()->route('admin.dashboard'),
        default => redirect()->route('home'),
    };
}

<?php

namespace App\Mail;

use App\Models\User;
use Illuminate\Bus\Queueable;
use Illuminate\Mail\Mailable;
use Illuminate\Queue\SerializesModels;

class LoginOtpMail extends Mailable
{
    use Queueable, SerializesModels;

    public $user;
    public $otp;

    public function __construct(User $user, string $otp)
    {
        $this->user = $user;
        $this->otp = $otp;
    }

    public function build()
    {
        return $this->subject('Your PetCare+ Login OTP')
                    ->view('emails.login-otp');
    }
}
